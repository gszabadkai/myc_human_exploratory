**To check the H2 table** read-only in Positron (AD HOC; it reads the saved object and fits nothing new beyond one slope per cohort):

```r
suppressMessages(library(dplyr))
x  <- readRDS("results/e49_signature_and_fig4g.rds")
cr <- x$frame %>% filter(in_set) %>% group_by(cohort) %>%
  summarise(c_raw = unname(coef(lm(OX_gsva ~ PROLIF))[2]), .groups = "drop")
h2 <- x$signature_coefficients %>%
  filter(instrument == "gsva", estimator %in% c("clean", "M_a"), readout %in% c("S6", "S7")) %>%
  left_join(cr, by = "cohort") %>%
  transmute(cohort, estimator, readout, b_P = b_PR, b_OX = b_OX_m3,
            inherited = -c_raw * b_OX_m3, remainder = b_PR + c_raw * b_OX_m3)
h2 %>% print(n = Inf)
```
